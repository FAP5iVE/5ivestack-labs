  <script>
    const hero = document.getElementById("hero");
    const root = document.documentElement;

    // Subtle mouse parallax tilt
    hero.addEventListener("mousemove", (e) => {
      const rect = hero.getBoundingClientRect();
      const normX = (e.clientX - rect.left) / rect.width - 0.5;
      const normY = (e.clientY - rect.top) / rect.height - 0.5;

      // Keep rotation angles subtle (-6deg to +6deg)
      root.style.setProperty("--mouse-x", `${(normX * 12).toFixed(2)}deg`);
      root.style.setProperty("--mouse-y", `${(-normY * 10).toFixed(2)}deg`);
    });

    // Reset smoothly when cursor leaves hero
    hero.addEventListener("mouseleave", () => {
      root.style.setProperty("--mouse-x", "0deg");
      root.style.setProperty("--mouse-y", "0deg");
    });
  </script>
